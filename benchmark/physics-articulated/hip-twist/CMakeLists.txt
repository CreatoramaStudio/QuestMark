# Hip-twist test: 3-DOF hip/shoulder (swing + twist) on limb chains, Jolt vs PhysX. Library set-up is in ../CMakeLists.txt.
# The limb builder (limb.cpp, skeleton.cpp) and backend.h are used from ../chain-limb.
set(CHAIN_DIR ${CMAKE_CURRENT_SOURCE_DIR}/../chain-limb)
set(hip_libs "")
set(hip_defs "")
if(JOLT_DIR)
  add_library(hip_jolt STATIC jolt3.cpp)
  target_link_libraries(hip_jolt PUBLIC artic_jolt_dep)
  target_include_directories(hip_jolt PRIVATE ${CHAIN_DIR})
  list(APPEND hip_libs hip_jolt)
  list(APPEND hip_defs HIP_HAVE_JOLT)
endif()
if(PHYSX_DIR)
  add_library(hip_physx STATIC physx3.cpp)
  target_link_libraries(hip_physx PUBLIC artic_physx_dep)
  target_include_directories(hip_physx PRIVATE ${CHAIN_DIR})
  list(APPEND hip_libs hip_physx)
  list(APPEND hip_defs HIP_HAVE_PHYSX)
endif()

add_executable(hip_twist main.cpp ${CHAIN_DIR}/limb.cpp ${CHAIN_DIR}/skeleton.cpp)
target_include_directories(hip_twist PRIVATE ${CHAIN_DIR})
target_link_libraries(hip_twist PRIVATE artic_ufbx ${hip_libs})
target_compile_definitions(hip_twist PRIVATE _CRT_SECURE_NO_WARNINGS ${hip_defs})
artic_target_defaults(hip_twist)
